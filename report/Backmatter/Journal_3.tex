#include <stdio.h>
#include "escape.h"

int draw(int x,int y,int w,int h);
void cleanup(int sig);
int animation(int vx,int vy);

int main(){

    printf("\x1B[2J");
    gotoxy(1,1);
    draw(1,1,20,10);
    animation(1,1);
    cleanup(0);

    // for (int i =0;i<256;i++){

    return 0;
}

int draw(int x,int y,int w,int h){
    for (int py = y ; py < y+h ; py++){
        for (int px = x ; px < x+w ; px++){
            gotoxy(px, py);
            if ((px == x && py == y) ||
                (px == w && py == y) ||
                (px == w && py == h) ||
                (px == x && py == h)) {
                printf("+");
            }
            else if ((py < y+h && px == x) || (py < y+h && px == w)){
                printf("|");
            }
            else if (px < x+w && py == y){
                printf("-");
            }
            else if (py == h){
                printf("-");
            }
        }
    }
}

void cleanup(int sig) {
    printf("\x1b[?25h");
    printf("\x1b[0m");
    fflush(stdout);
    _exit(0);
}

int animation(int vx,int vy){
    int x = 10;
    int y = 5;
    int oldx = x;
    int oldy = y;
    int n = 0;

    signal(SIGINT, cleanup);

    while (1){

        printf("\x1b[%d;%dH ", oldy, oldx);

        oldx = x;
        oldy = y;

        fflush(stdout);

        x += vx;
        y += vy;

        printf("\x1b[%d;%dH", y, x);
        printf("\x1b[31mO\x1b[0m");

        if (x <= 1 || x >= 20) vx = -vx;
        if (y <= 1 || y >= 10) vy = -vy;

        usleep(30000);

        if (y >= 10){
            printf("\n\n\nnumber of bounces: %d", n++);
        }
    }
}
